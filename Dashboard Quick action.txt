import React from 'react';
import { Link } from 'react-router-dom';
import { motion } from 'framer-motion';
import { ArrowRight } from 'lucide-react';

const gradients = {
  'bg-emerald-500': 'linear-gradient(135deg, #10b981, #059669)',
  'bg-sky-500': 'linear-gradient(135deg, #0ea5e9, #0284c7)',
  'bg-amber-500': 'linear-gradient(135deg, #f59e0b, #d97706)',
  'bg-violet-500': 'linear-gradient(135deg, #8b5cf6, #7c3aed)',
  'bg-rose-500': 'linear-gradient(135deg, #f43f5e, #e11d48)',
};

const glows = {
  'bg-emerald-500': 'rgba(16,185,129,0.15)',
  'bg-sky-500': 'rgba(14,165,233,0.15)',
  'bg-amber-500': 'rgba(245,158,11,0.15)',
  'bg-violet-500': 'rgba(139,92,246,0.15)',
  'bg-rose-500': 'rgba(244,63,94,0.15)',
};

export default function QuickAction({ icon: Icon, title, description, path, color, delay }) {
  return (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ delay: delay * 0.08, type: 'spring', stiffness: 250 }}
      whileHover={{ y: -6, scale: 1.01 }}
      whileTap={{ scale: 0.98 }}
    >
      <Link to={path}>
        <div
          className="group relative overflow-hidden rounded-2xl border border-white/60 p-6 cursor-pointer transition-all duration-300"
          style={{
            background: 'rgba(255,255,255,0.8)',
            backdropFilter: 'blur(10px)',
            boxShadow: `0 2px 16px ${glows[color] || 'rgba(0,0,0,0.05)'}`,
          }}
          onMouseEnter={e => {
            e.currentTarget.style.boxShadow = `0 12px 32px ${glows[color] || 'rgba(0,0,0,0.1)'}`;
          }}
          onMouseLeave={e => {
            e.currentTarget.style.boxShadow = `0 2px 16px ${glows[color] || 'rgba(0,0,0,0.05)'}`;
          }}
        >
          {/* Top accent line */}
          <div
            className="absolute top-0 left-0 right-0 h-1 rounded-t-2xl opacity-80 group-hover:opacity-100 transition-opacity"
            style={{ background: gradients[color] || color }}
          />

          {/* Floating bg circle */}
          <div
            className="absolute -bottom-4 -right-4 w-20 h-20 rounded-full opacity-0 group-hover:opacity-10 transition-opacity duration-300"
            style={{ background: gradients[color] || color }}
          />

          <motion.div
            whileHover={{ rotate: 5, scale: 1.1 }}
            className="w-14 h-14 rounded-2xl flex items-center justify-center mb-4 shadow-md"
            style={{ background: gradients[color] || color }}
          >
            <Icon className="w-7 h-7 text-white" />
          </motion.div>

          <h3 className="text-base font-bold text-foreground mb-1">{title}</h3>
          <p className="text-sm text-muted-foreground leading-relaxed mb-4">{description}</p>

          <div
            className="flex items-center text-sm font-semibold gap-1 opacity-0 group-hover:opacity-100 transition-all duration-200 translate-y-1 group-hover:translate-y-0"
            style={{ color: color.replace('bg-', '').includes('emerald') ? '#10b981' :
              color.includes('sky') ? '#0ea5e9' :
              color.includes('amber') ? '#f59e0b' :
              color.includes('violet') ? '#8b5cf6' : '#f43f5e' }}
          >
            <span>Open</span>
            <motion.div animate={{ x: [0, 4, 0] }} transition={{ repeat: Infinity, duration: 1.2 }}>
              <ArrowRight className="w-4 h-4" />
            </motion.div>
          </div>
        </div>
      </Link>
    </motion.div>
  );
}
